# 동래아 공동육아 · 익명게시판 설정 (처음 한 번만)

사이트: 동래아 공동육아 홈페이지 — 익명게시판 `board.html`, 모임 요청 `meet.html` (홈에도 다가오는 모임이 보임)
주소: https://kimddubuck.github.io/coparenting/ (Settings → Pages에서 main 브랜치를 켜야 열림)

## 글이 저장되는 곳
- 성장노트와 같은 Firebase 프로젝트(yunhaeducation)의 Firestore, `coparenting_opinions` 컬렉션
- 익명게시판(`board.html`) 주제: habit(① 배웠으면 하는 행동), dev(② 해 봤으면 하는 경험), free(💬 자유)
- 🆘 SOS 예약(공동육아 예약 도우미): `coparenting_sos/날짜` 문서의 h9~h18 (그 날 그 시간 SOS 예약 수). 예약 +1, 취소 -1 (0 아래 불가). 규칙은 h8~h20까지 허용
- 놀이 추가(`play.html`): `coparenting_plays` 컬렉션 — 이름, ①/② 구분, 종류, 건네는 말, 하는 방법, 준비물
- 모임(`meet.html`): topic `meet` 글 + 주최자 별명(host, 20자 이내, 공개됨). 연 휴대폰에서 '모임 취소' → cancelled: true (카드에 취소 표시) + 날짜·시간(9시~18시, 예전 글은 오전/점심/오후/저녁) + 참석(joins)·미확정(maybes)·불참(nos) 수, 댓글은 글 아래 `comments` 컬렉션
- 이름·로그인 정보는 저장하지 않음. 누구나 읽을 수 있음 (사이트가 공개라서)
- 참석·미확정·불참은 같은 기기에서 셋 중 하나만 한 번 누르게 했지만, 다른 기기나 브라우저로는 다시 누를 수 있음

## 1단계: Firestore 보안 규칙 (전체를 이걸로 바꾸기)
Firebase 콘솔 → Firestore Database → 규칙 탭 → 전체 지우고 붙여 넣기 → 게시

```
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // 성장노트: 로그인한 사람만 읽고 쓰기 (기존 그대로)
    match /{document=**} {
      allow read, write: if request.auth != null;
    }

    // 동래아 공동육아: 익명게시판 + 모임 요청
    match /coparenting_opinions/{id} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['topic', 'text', 'play', 'date', 'slot', 'host', 'joins', 'maybes', 'nos', 'createdAt'])
        && request.resource.data.topic in ['habit', 'dev', 'meet', 'free']
        && request.resource.data.text is string
        && request.resource.data.text.size() > 0
        && request.resource.data.text.size() <= 500
        && (!('play' in request.resource.data)
            || (request.resource.data.play is string && request.resource.data.play.size() <= 40))
        && (!('date' in request.resource.data)
            || (request.resource.data.date is string && request.resource.data.date.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')))
        && (!('slot' in request.resource.data)
            || request.resource.data.slot in ['오전', '점심', '오후', '저녁', '9시', '10시', '11시', '12시', '13시', '14시', '15시', '16시', '17시', '18시'])
        && (!('host' in request.resource.data)
            || (request.resource.data.host is string && request.resource.data.host.size() > 0 && request.resource.data.host.size() <= 20))
        && (!('joins' in request.resource.data) || request.resource.data.joins == 0)
        && (!('maybes' in request.resource.data) || request.resource.data.maybes == 0)
        && (!('nos' in request.resource.data) || request.resource.data.nos == 0)
        && request.resource.data.createdAt == request.time;
      // 참석(joins)·미확정(maybes)·불참(nos): 셋 중 하나의 숫자를 1씩 올리기
      // 모임 취소: cancelled 를 true 로 한 번만 바꾸기 (되돌릴 수 없음)
      allow update: if (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['cancelled'])
                        && request.resource.data.cancelled == true && resource.data.topic == 'meet')
                    || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['joins'])
                        && request.resource.data.joins == resource.data.get('joins', 0) + 1)
                    || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['maybes'])
                        && request.resource.data.maybes == resource.data.get('maybes', 0) + 1)
                    || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['nos'])
                        && request.resource.data.nos == resource.data.get('nos', 0) + 1);
      allow delete: if request.auth != null;

      // 모임 요청 댓글: 누구나 읽고 쓰기, 수정 불가, 삭제는 로그인한 사람만
      match /comments/{cid} {
        allow read: if true;
        allow create: if request.resource.data.keys().hasOnly(['text', 'createdAt'])
          && request.resource.data.text is string
          && request.resource.data.text.size() > 0
          && request.resource.data.text.size() <= 300
          && request.resource.data.createdAt == request.time;
        allow update: if false;
        allow delete: if request.auth != null;
      }
    }

    // 🆘 SOS 예약: 날짜별 문서에 시간(h8~h20) 하나만 1씩 올리기(예약) 또는 내리기(취소, 0 아래로는 안 됨)
    match /coparenting_sos/{day} {
      allow read: if true;
      allow create: if day.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')
        && ((request.resource.data.keys().hasOnly(['h8']) && request.resource.data.h8 == 1)
        || (request.resource.data.keys().hasOnly(['h9']) && request.resource.data.h9 == 1)
        || (request.resource.data.keys().hasOnly(['h10']) && request.resource.data.h10 == 1)
        || (request.resource.data.keys().hasOnly(['h11']) && request.resource.data.h11 == 1)
        || (request.resource.data.keys().hasOnly(['h12']) && request.resource.data.h12 == 1)
        || (request.resource.data.keys().hasOnly(['h13']) && request.resource.data.h13 == 1)
        || (request.resource.data.keys().hasOnly(['h14']) && request.resource.data.h14 == 1)
        || (request.resource.data.keys().hasOnly(['h15']) && request.resource.data.h15 == 1)
        || (request.resource.data.keys().hasOnly(['h16']) && request.resource.data.h16 == 1)
        || (request.resource.data.keys().hasOnly(['h17']) && request.resource.data.h17 == 1)
        || (request.resource.data.keys().hasOnly(['h18']) && request.resource.data.h18 == 1)
        || (request.resource.data.keys().hasOnly(['h19']) && request.resource.data.h19 == 1)
        || (request.resource.data.keys().hasOnly(['h20']) && request.resource.data.h20 == 1));
      allow update: if (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['h8']) && (request.resource.data.h8 == resource.data.get('h8', 0) + 1 || (resource.data.get('h8', 0) >= 1 && request.resource.data.h8 == resource.data.get('h8', 0) - 1)))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['h9']) && (request.resource.data.h9 == resource.data.get('h9', 0) + 1 || (resource.data.get('h9', 0) >= 1 && request.resource.data.h9 == resource.data.get('h9', 0) - 1)))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['h10']) && (request.resource.data.h10 == resource.data.get('h10', 0) + 1 || (resource.data.get('h10', 0) >= 1 && request.resource.data.h10 == resource.data.get('h10', 0) - 1)))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['h11']) && (request.resource.data.h11 == resource.data.get('h11', 0) + 1 || (resource.data.get('h11', 0) >= 1 && request.resource.data.h11 == resource.data.get('h11', 0) - 1)))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['h12']) && (request.resource.data.h12 == resource.data.get('h12', 0) + 1 || (resource.data.get('h12', 0) >= 1 && request.resource.data.h12 == resource.data.get('h12', 0) - 1)))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['h13']) && (request.resource.data.h13 == resource.data.get('h13', 0) + 1 || (resource.data.get('h13', 0) >= 1 && request.resource.data.h13 == resource.data.get('h13', 0) - 1)))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['h14']) && (request.resource.data.h14 == resource.data.get('h14', 0) + 1 || (resource.data.get('h14', 0) >= 1 && request.resource.data.h14 == resource.data.get('h14', 0) - 1)))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['h15']) && (request.resource.data.h15 == resource.data.get('h15', 0) + 1 || (resource.data.get('h15', 0) >= 1 && request.resource.data.h15 == resource.data.get('h15', 0) - 1)))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['h16']) && (request.resource.data.h16 == resource.data.get('h16', 0) + 1 || (resource.data.get('h16', 0) >= 1 && request.resource.data.h16 == resource.data.get('h16', 0) - 1)))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['h17']) && (request.resource.data.h17 == resource.data.get('h17', 0) + 1 || (resource.data.get('h17', 0) >= 1 && request.resource.data.h17 == resource.data.get('h17', 0) - 1)))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['h18']) && (request.resource.data.h18 == resource.data.get('h18', 0) + 1 || (resource.data.get('h18', 0) >= 1 && request.resource.data.h18 == resource.data.get('h18', 0) - 1)))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['h19']) && (request.resource.data.h19 == resource.data.get('h19', 0) + 1 || (resource.data.get('h19', 0) >= 1 && request.resource.data.h19 == resource.data.get('h19', 0) - 1)))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['h20']) && (request.resource.data.h20 == resource.data.get('h20', 0) + 1 || (resource.data.get('h20', 0) >= 1 && request.resource.data.h20 == resource.data.get('h20', 0) - 1)));
      allow delete: if request.auth != null;
    }

    // 놀이 추가: 누구나 놀이를 올리기, 수정 불가, 삭제는 로그인한 사람만
    match /coparenting_plays/{id} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['name', 'kind', 'cat', 'say', 'how', 'items', 'createdAt'])
        && request.resource.data.name is string
        && request.resource.data.name.size() > 0
        && request.resource.data.name.size() <= 30
        && request.resource.data.kind in ['habit', 'dev']
        && (!('cat' in request.resource.data)
            || (request.resource.data.cat is string && request.resource.data.cat.size() <= 20))
        && (!('say' in request.resource.data)
            || (request.resource.data.say is string && request.resource.data.say.size() <= 40))
        && request.resource.data.how is string
        && request.resource.data.how.size() > 0
        && request.resource.data.how.size() <= 200
        && request.resource.data.items is list
        && request.resource.data.items.size() <= 10
        && request.resource.data.createdAt == request.time;
      allow update: if false;
      allow delete: if request.auth != null;
    }
  }
}
```
